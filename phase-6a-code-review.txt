    return response.data;
  },

    const response = await apiClient.get<ApiSuccess<ApiDailyOverview>>('/habits/daily-overview', { params: { date } });
    return response.data.data;
  },
  async get(uuid: string): Promise<ApiHabit> {
    const response = await apiClient.get<ApiSuccess<ApiHabit>>(`/habits/${uuid}`);
    return response.data.data;
  },
  async create(request: SaveHabitRequest): Promise<ApiHabit> {
    const response = await apiClient.post<ApiSuccess<ApiHabit>>('/habits', request);
    return response.data.data;
  },
  async update(uuid: string, request: SaveHabitRequest): Promise<ApiHabit> {
    const response = await apiClient.put<ApiSuccess<ApiHabit>>(`/habits/${uuid}`, request);
    return response.data.data;
  },
  async archive(uuid: string): Promise<ApiHabit> {
    const response = await apiClient.post<ApiSuccess<ApiHabit>>(`/habits/${uuid}/archive`);
    return response.data.data;
  },
  async restore(uuid: string): Promise<ApiHabit> {
    const response = await apiClient.post<ApiSuccess<ApiHabit>>(`/habits/${uuid}/restore`);
    return response.data.data;
  },
  async remove(uuid: string): Promise<void> {
    await apiClient.delete(`/habits/${uuid}`);
  },
  async logs(uuid: string, filters: HabitLogFilters): Promise<ApiPaginatedResponse<ApiHabitLog>> {

import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { habitApi } from '@/services/api/habitApi';
import type { HabitFilters, HabitLogFilters, SaveHabitLogRequest, SaveHabitRequest } from '@/services/api/contracts';

export const habitKeys = {
  all: ['habits'] as const,
  lists: () => [...habitKeys.all, 'list'] as const,
  list: (filters: HabitFilters) => [...habitKeys.lists(), filters] as const,
  detail: (uuid: string) => [...habitKeys.all, 'detail', uuid] as const,
  overview: (date: string) => [...habitKeys.all, 'overview', date] as const,
  logs: (uuid: string, filters: HabitLogFilters) => [...habitKeys.all, 'logs', uuid, filters] as const,
};

export function useHabits(filters: HabitFilters, enabled = true) {
  return useQuery({ queryKey: habitKeys.list(filters), queryFn: () => habitApi.list(filters), enabled });
}

export function useDailyOverview(date: string, enabled = true) {
  return useQuery({ queryKey: habitKeys.overview(date), queryFn: () => habitApi.dailyOverview(date), enabled });
}

export function useHabit(uuid: string) {
  return useQuery({ queryKey: habitKeys.detail(uuid), queryFn: () => habitApi.get(uuid), enabled: Boolean(uuid) });
}

export function useHabitLogs(uuid: string, filters: HabitLogFilters, enabled = true) {
  return useQuery({ queryKey: habitKeys.logs(uuid, filters), queryFn: () => habitApi.logs(uuid, filters), enabled: Boolean(uuid) && enabled });
}

export function useSaveHabit(uuid?: string) {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (request: SaveHabitRequest) => (uuid ? habitApi.update(uuid, request) : habitApi.create(request)),
    onSuccess: async (habit) => {
      await queryClient.invalidateQueries({ queryKey: habitKeys.all });
      queryClient.setQueryData(habitKeys.detail(habit.uuid), habit);
    },
  });
}

export function useHabitAction(action: 'archive' | 'restore' | 'delete') {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async (uuid: string) => {
      if (action === 'archive') return habitApi.archive(uuid);
      if (action === 'restore') return habitApi.restore(uuid);
      await habitApi.remove(uuid);
      return null;
    },
    onSuccess: async () => {
      if (action === 'delete') {
        await queryClient.invalidateQueries({ queryKey: habitKeys.all, refetchType: 'none' });
        return;
      }
      await queryClient.invalidateQueries({ queryKey: habitKeys.all });
    },
  });
}

export function useLogHabit(uuid: string) {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (request: SaveHabitLogRequest) => habitApi.logProgress(uuid, request),
    onSuccess: async () => {
      await queryClient.invalidateQueries({ queryKey: habitKeys.all });
    },
  });
}

export function useRemoveHabitLog(uuid: string) {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (date: string) => habitApi.removeLog(uuid, date),
    onSuccess: async () => {
      await queryClient.invalidateQueries({ queryKey: habitKeys.all });
    },
  });
}

import { z } from 'zod';

export const HABIT_TARGET_DAYS = ['mon', 'tue', 'wed', 'thu', 'fri', 'sat', 'sun'] as const;

export const habitFormSchema = z
  .object({
    title: z.string().trim().min(1, 'Enter a habit title.').max(255, 'Use 255 characters or fewer.'),
    description: z.string().max(1000, 'Use 1000 characters or fewer.'),
    frequency: z.enum(['daily', 'custom']),
    target_days: z.array(z.enum(HABIT_TARGET_DAYS)),
    color: z.string().regex(/^#[0-9a-fA-F]{6}$/, 'Choose a valid color.'),
  })
  .refine((values) => values.frequency !== 'custom' || values.target_days.length > 0, {
    message: 'Custom habits require at least one weekday.',
    path: ['target_days'],
  });

export type HabitFormValues = z.infer<typeof habitFormSchema>;

export function habitTargetDayLabel(day: (typeof HABIT_TARGET_DAYS)[number]): string {
  return day.charAt(0).toUpperCase() + day.slice(1);
}

export function habitErrorMessage(error: unknown): string {
  if (typeof error === 'object' && error !== null && 'status' in error) {
    const status = error.status;
    if (status === 401) return 'Your session has expired. Sign in again to continue.';
    if (status === 403) return 'You no longer have permission to manage habits.';
    if (status === 404) return 'This habit is unavailable or outside your access scope.';
    if (status === 422) {
      if ('errors' in error && typeof error.errors === 'object' && error.errors !== null) {
        const firstMessage = Object.values(error.errors).find((value) => typeof value === 'string');
        if (typeof firstMessage === 'string') return firstMessage;
      }
      return 'The habit details were rejected. Review the fields and try again.';
    }
    if (status === 429) return 'Too many requests. Wait a moment and try again.';
  }
  return 'Hirgal could not complete this request. Check your connection and try again.';
}

function localCalendarDate(date: Date): string {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const day = String(date.getDate()).padStart(2, '0');
  return `${year}-${month}-${day}`;
}

function todayForUser(timeZone?: string): string {
  try {
    return new Intl.DateTimeFormat('en-CA', {
      timeZone: timeZone ?? 'UTC',
      year: 'numeric',
      month: '2-digit',
      day: '2-digit',
    }).format(new Date());
  } catch {
    return localCalendarDate(new Date());
  }
}

function addDays(isoDate: string, days: number): string {
  const [year, month, day] = isoDate.split('-').map(Number);
  const shifted = new Date(Date.UTC(year ?? 1970, (month ?? 1) - 1, (day ?? 1) + days));
  return shifted.toISOString().slice(0, 10);
}

function CheckInRow({ date, item }: { date: string; item: ApiDailyOverviewItem }) {
  const { habit, is_scheduled, log } = item;
  const logHabit = useLogHabit(habit.uuid);
  const removeHabitLog = useRemoveHabitLog(habit.uuid);
  const [confirmRemoval, setConfirmRemoval] = useState(false);
  const [actionError, setActionError] = useState<string | null>(null);

  async function record(status: HabitLogStatus): Promise<void> {
    setActionError(null);
    try {
      await logHabit.mutateAsync({ logged_date: date, status });
    } catch (error) {
      setActionError(habitErrorMessage(error));
    }
  }

  async function confirmRemove(): Promise<void> {
    setActionError(null);
    try {
      await removeHabitLog.mutateAsync(date);
      setConfirmRemoval(false);
    } catch (error) {
      setActionError(habitErrorMessage(error));
    }
  }

  const badge = log ? logStatusBadges[log.status] : null;

  return (
    <li className="flex flex-wrap items-center justify-between gap-3 py-3">
      <div className="flex min-w-0 items-center gap-3">
        <span
          aria-hidden="true"
          className="size-3 shrink-0 rounded-full"
          style={{ backgroundColor: habit.color ?? '#94a3b8' }}
        />
        <div className="min-w-0">
          <Link
            className="text-sm font-medium hover:text-primary focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
            to={`/habits/${habit.uuid}`}
          >
            {habit.title}
          </Link>
          <p className="text-xs text-muted-foreground">
            {!is_scheduled
              ? 'Not scheduled on this date'
              : badge
                ? `Checked in for ${date}`
                : 'Scheduled for this date'}
          </p>
        </div>
      </div>
      <div className="flex items-center gap-2">
        {!is_scheduled ? null : badge ? (
          <>
            <StatusBadge label={badge.label} status={badge.status} />
            <Button
              aria-label={`Remove check-in for ${habit.title}`}
              disabled={removeHabitLog.isPending}

              variant="outline"

import { useEffect, useState, type ReactNode } from 'react';

import { useHabit, useSaveHabit } from '../habit-queries';
import {

  habitTargetDayLabel,
  type HabitFormValues,
} from '../habit-validation';
import type { SaveHabitRequest } from '@/services/api/contracts';

export function HabitFormRoute() {
  const { habitUuid } = useParams();
  const navigate = useNavigate();
  const { user } = useAuth();
  const existing = useHabit(habitUuid ?? '');
  const save = useSaveHabit(habitUuid);
  const [errors, setErrors] = useState<Record<string, string>>({});
  const form = useForm<HabitFormValues>({
    defaultValues: {
      title: '',
      description: '',
      frequency: 'daily',
      target_days: [],
      color: '#3B82F6',
    },
  });
  const seeded = existing.data;
  const resetForm = form.reset;
  const frequency = useWatch({ control: form.control, name: 'frequency' });
  const targetDays = useWatch({ control: form.control, name: 'target_days' });

  useEffect(() => {
    if (seeded) {
      resetForm({
        title: seeded.title,
        description: seeded.description ?? '',
        frequency: seeded.frequency,
        target_days: seeded.target_days ?? [],
        color: seeded.color ?? '#3B82F6',
      });
    }
  }, [resetForm, seeded]);

  if (habitUuid && existing.isPending) {
    return (
      <div className="flex min-h-48 items-center justify-center">
        <LoadingState label="Loading habit" />
      </div>
    );
  }
  if (habitUuid && existing.isError) {
    return (
      <ErrorState
        description={habitErrorMessage(existing.error)}
        onRetry={() => void existing.refetch()}
        title="Habit unavailable"
      />
    );
  }
  if (habitUuid && !seeded) {
    return <ErrorState description="This habit is unavailable." title="Habit unavailable" />;
  }
  if (!hasPermission(user, 'habits.manage')) {
    return (
      <ErrorState
        description={
          habitUuid
            ? 'Editing habits requires the habits manage permission.'
            : 'Your account cannot create habits.'
        }
        title={habitUuid ? 'Editing unavailable' : 'Creation unavailable'}
      />
    );
  }

  async function submit(values: HabitFormValues): Promise<void> {
    const parsed = habitFormSchema.safeParse(values);
    if (!parsed.success) {
      setErrors(
        Object.fromEntries(
          parsed.error.issues.map((issue) => [String(issue.path[0]), issue.message]),
        ),
      );
      return;
    }
    setErrors({});
    const request: SaveHabitRequest = {
      title: parsed.data.title,
      description: parsed.data.description || null,
      frequency: parsed.data.frequency,
      target_days: parsed.data.target_days,
      color: parsed.data.color,
    };
    try {
      const habit = await save.mutateAsync(request);
      navigate(`/habits/${habit.uuid}`);
    } catch {
      return;
    }
  }

  return (
    <section className="mx-auto max-w-3xl space-y-6">
      <header>
        <Link
          className="text-sm text-primary hover:underline"
          to={habitUuid ? `/habits/${habitUuid}` : '/habits'}
        >
          {habitUuid ? 'Back to habit' : 'Habits'}
        </Link>
        <h1 className="mt-2 text-2xl font-semibold tracking-tight">
          {habitUuid ? 'Edit habit' : 'Create habit'}
        </h1>
        <p className="mt-2 text-sm text-muted-foreground">
          {habitUuid
            ? 'Update the habit details supported by the habits API.'
            : 'Define a habit and choose when it should be tracked.'}
        </p>
      </header>
      <Card>
        <CardContent className="pt-6">
          <form
            className="space-y-5"
            noValidate
            onSubmit={form.handleSubmit(submit)}
          >
            <Field error={errors.title} label="Habit title">
              <Input {...form.register('title')} autoComplete="off" />
            </Field>
            <Field error={errors.description} label="Description">
              <Textarea {...form.register('description')} rows={4} />
            </Field>
            <div className="grid gap-4 sm:grid-cols-2">
              <Field error={errors.frequency} label="Frequency">
                <select
                  className="h-10 w-full rounded-md border bg-background px-3"
                  {...form.register('frequency')}
                >
                  <option value="daily">Daily</option>
                  <option value="custom">Custom days</option>
                </select>
              </Field>
              <Field error={errors.color} label="Color">
                <Input defaultValue="#3B82F6" type="color" {...form.register('color')} />
              </Field>
            </div>
            {frequency === 'custom' ? (
              <div className="space-y-1.5">
                <p className="text-sm font-medium">
                  Target days
                  <span aria-hidden="true" className="ml-1 text-destructive">
                    *
                  </span>
                </p>
                <div className="flex flex-wrap gap-3">
                  {HABIT_TARGET_DAYS.map((day) => (
                    <label className="flex items-center gap-2 text-sm font-medium" key={day}>
                      <Checkbox
                        aria-label={habitTargetDayLabel(day)}
                        checked={targetDays.includes(day)}
                        onCheckedChange={(checked) => {
                          const next =
                            checked === true
                              ? [...targetDays, day]
                              : targetDays.filter((value) => value !== day);
                          form.setValue('target_days', next, { shouldValidate: true });
                        }}
                      />
                      {habitTargetDayLabel(day)}
                    </label>
                  ))}
                </div>
                {errors.target_days ? (
                  <p className="text-xs text-destructive" role="alert">
                    {errors.target_days}
                  </p>
                ) : null}
              </div>
            ) : null}
            {save.isError ? (
              <p className="text-sm text-destructive" role="alert">
                {habitErrorMessage(save.error)}
              </p>
            ) : null}
            <div className="flex gap-3">
              <Button loading={save.isPending} type="submit">
                {habitUuid ? 'Save changes' : 'Create habit'}
              </Button>
              <Button asChild variant="outline">
                <Link to={habitUuid ? `/habits/${habitUuid}` : '/habits'}>Cancel</Link>
              </Button>
            </div>
          </form>
        </CardContent>
      </Card>
    </section>
  );
}

function Field({
  label,
  error,
  children,
}: {
  label: string;
  error: string | undefined;
  children: ReactNode;
}) {
  return (
    <label className="block space-y-1.5 text-sm font-medium">
      {label}
      {children}
      {error ? (
        <span className="block text-xs font-normal text-destructive" role="alert">
          {error}
        </span>
      ) : null}
    </label>
  );
}
